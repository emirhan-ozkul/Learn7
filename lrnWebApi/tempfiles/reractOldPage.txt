
import { useState } from "react";

interface TranslationResponse {
    english: string;
    turkish: string;
    grammar: string;
    level: string;
}

function ApiTest() {
    const [topic, setTopic] = useState("");
    const [level, setLevel] = useState("");

    const [result, setResult] = useState<TranslationResponse | null>(null);

    const [loading, setLoading] = useState(false);

    const [error, setError] = useState<string | null>(null);

    // --------------------------------------------------
    // POST /api/Translation/generate
    // --------------------------------------------------

    const generateSentence = async () => {
        setLoading(true);
        setError(null);
        setResult(null);

        try {
            const response = await fetch(
                "http://192.168.1.50:5000/api/Translation/generate",
                {
                    method: "POST",

                    headers: {
                        "Content-Type": "application/json",
                    },

                    body: JSON.stringify({
                        Topic: topic,
                        Level: level
                    }),
                }
            );

            // HTTP 400 / 500 / 429 vb.
            if (!response.ok) {
                const errorText = await response.text();

                throw new Error(
                    `HTTP ${response.status}: ${errorText}`
                );
            }

            const data: TranslationResponse =
                await response.json();

            console.log("POST Response:", data);

            setResult(data);
        } catch (err) {
            console.error("POST Error:", err);

            if (err instanceof Error) {
                setError(err.message);
            } else {
                setError("Bilinmeyen bir hata oluştu.");
            }
        } finally {
            setLoading(false);
        }
    };

    // --------------------------------------------------
    // GET /api/Translation/getR
    // --------------------------------------------------

    const getLastResult = async () => {
        setLoading(true);
        setError(null);
        setResult(null);

        try {
            const response = await fetch(
                "http://192.168.1.50:5000/api/Translation/getR"
            );

            if (!response.ok) {
                const errorText = await response.text();

                throw new Error(
                    `HTTP ${response.status}: ${errorText}`
                );
            }

            const data: TranslationResponse =
                await response.json();

            console.log("GET Response:", data);

            setResult(data);
        } catch (err) {
            console.error("GET Error:", err);

            if (err instanceof Error) {
                setError(err.message);
            } else {
                setError("Bilinmeyen bir hata oluştu.");
            }
        } finally {
            setLoading(false);
        }
    };

    return (
        <div
            style={{
                minHeight: "100vh",
                padding: "40px",
                backgroundColor: "#1e1e1e",
                color: "white",
                fontFamily: "Arial",
            }}
        >
            <div
                style={{
                    maxWidth: "800px",
                    margin: "0 auto",
                }}
            >
                <h1>Translation API Test</h1>

                <p>
                    Bu sayfa C# backend API'sini test etmek için
                    oluşturulmuştur.
                </p>

                <hr />

                {/* PROMPT */}

                <h2>POST Test</h2>

                <label>
                    Prompt:
                </label>

                <textarea
                    value={topic}
                    onChange={(e) =>
                        setTopic(e.target.value)
                    }
                    rows={1}
                    style={{
                        width: "100%",
                        marginTop: "10px",
                        padding: "10px",
                        fontSize: "16px",
                        boxSizing: "border-box",
                    }}
                />
                <textarea
                    value={level}
                    onChange={(e) =>
                        setLevel(e.target.value)
                    }
                    rows={1}
                    style={{
                        width: "100%",
                        marginTop: "10px",
                        padding: "10px",
                        fontSize: "16px",
                        boxSizing: "border-box",
                    }}
                />

                <br />
                <br />

                <button
                    onClick={generateSentence}
                    disabled={loading}
                    style={{
                        padding: "12px 20px",
                        fontSize: "16px",
                        cursor: loading
                            ? "not-allowed"
                            : "pointer",
                    }}
                >
                    {loading
                        ? "İstek Gönderiliyor..."
                        : "POST - Yeni Cümle Oluştur"}
                </button>

                <br />
                <br />

                {/* GET */}

                <h2>GET Test</h2>

                <button
                    onClick={getLastResult}
                    disabled={loading}
                    style={{
                        padding: "12px 20px",
                        fontSize: "16px",
                        cursor: loading
                            ? "not-allowed"
                            : "pointer",
                    }}
                >
                    GET - Sonucu Getir
                </button>

                <br />
                <br />

                {/* ERROR */}

                {error && (
                    <div
                        style={{
                            padding: "15px",
                            backgroundColor: "#5c1f1f",
                            border: "1px solid #ff5555",
                            borderRadius: "5px",
                            marginTop: "20px",
                        }}
                    >
                        <strong>Hata:</strong>

                        <pre
                            style={{
                                whiteSpace: "pre-wrap",
                            }}
                        >
                            {error}
                        </pre>
                    </div>
                )}

                {/* RESULT */}

                {result && (
                    <div
                        style={{
                            marginTop: "30px",
                            padding: "20px",
                            backgroundColor: "#292929",
                            borderRadius: "10px",
                        }}
                    >
                        <h2>API Response</h2>

                        <div
                            style={{
                                marginTop: "20px",
                            }}
                        >
                            <strong>English:</strong>

                            <p>
                                {result.english}
                            </p>
                        </div>

                        <div>
                            <strong>Turkish:</strong>

                            <p>
                                {result.turkish}
                            </p>
                        </div>

                        <div>
                            <strong>Grammar:</strong>

                            <p>
                                {result.grammar}
                            </p>
                        </div>

                        <div>
                            <strong>Level:</strong>

                            <p>
                                {result.level}
                            </p>
                        </div>

                        <hr />

                        <h3>Raw JSON</h3>

                        <pre
                            style={{
                                backgroundColor: "#111",
                                padding: "15px",
                                overflowX: "auto",
                            }}
                        >
                            {JSON.stringify(
                                result,
                                null,
                                2
                            )}
                        </pre>
                    </div>
                )}
            </div>
        </div>
    );
}

export default ApiTest;
