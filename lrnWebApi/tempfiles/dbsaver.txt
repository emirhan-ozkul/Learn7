using System;
using System.Diagnostics.Metrics;
using lrnWebApi.Database;
using lrnWebApi.Models;
using lrnWebApi.SettingCS;
using Microsoft.Data.Sqlite;
using SQLitePCL;

namespace lrnWebApi.Services;

public class InitializeDatabase
{
    public async static void CreateInitializeDatabase()
    {
        string CreateTableINE = @"CREATE TABLE IF NOT EXISTS sentences (Id INTEGER PRIMARY KEY AUTOINCREMENT, english TEXT, Renglish TEXT, turkish TEXT, grammar TEXT, topic TEXT, level TEXT, ısUsed INTEGER CHECK (ısUsed IN (0, 1)) DEFAULT 0);";
        string InsertCTINETable = @"INSERT INTO sentences (english, Renglish, turkish, grammar, level) VALUES ('I watched TV yesterday.','tv yesterday watched ı','Dün televizyon izledim.','Simple Past - Irregular','A1');";
        string CreatesUserTable = @"CREATE TABLE IF NOT EXISTS users (Id INTEGER PRIMARY KEY AUTOINCREMENT, username TEXT, email TEXT, passwordhash TEXT, createdAt TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP, lastlogin TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP)";
        string UserAttempt = @"CREATE TABLE IF NOT EXISTS userAttept (ID INTEGER PRIMARY KEY AUTOINCREMENT, userID INTEGER, contentID INTEGER, userAnswer TEXT, IsCorrect INTEGER CHECK(IsCorrect IN (0, 1)), Solving TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP)";
        string InsterCUTable = @"INSERT INTO users (username, email, passwordhash) VALUES ('admin','admin','admin');";
        string CheckUser= @"SELECT COUNT(*) FROM users; " + @"SELECT COUNT(*) FROM sentences";
        string CheckContent = @"SELECT COUNT(*) FROM sentences";

        using (var conn = new SqliteConnection(DatabaseSettings.ConnectionString))
        {
            conn.Open();
            using (var transaction = conn.BeginTransaction())
            {
                try
                {
                    using (var varUAT = new SqliteCommand(UserAttempt, conn, transaction))
                    {
                        varUAT.ExecuteNonQuery();
                    }
                    using (var commandCheck = new SqliteCommand(CreatesUserTable, conn, transaction))
                    {
                        commandCheck.ExecuteNonQuery();
                        using(var checkUserQr = new SqliteCommand(CheckUser, conn, transaction))
                        {
                            int chuserqr = Convert.ToInt32(checkUserQr.ExecuteScalar());
                            if (chuserqr == 0)
                            {
                                using(var insertUser = new SqliteCommand(InsterCUTable, conn, transaction))
                                {
                                    Console.WriteLine("Kullanıcı verileri eklendi");
                                    insertUser.ExecuteNonQuery();
                                }
                            }
                        }
                    }
                    using (var commandCheck = new SqliteCommand(CreateTableINE, conn, transaction))
                    {
                        commandCheck.ExecuteNonQuery();
                        using(var checkContentQr = new SqliteCommand(CheckContent, conn, transaction))
                        {
                            int chcontentqr = Convert.ToInt32(checkContentQr.ExecuteScalar());
                            if (chcontentqr == 0)
                            {
                                using(var insertUser = new SqliteCommand(InsertCTINETable, conn, transaction))
                                {
                                    Console.WriteLine("Cümle verileri eklendi");
                                    insertUser.ExecuteNonQuery();
                                }
                            }
                        }
                    }
                    
                    
                }
                catch(Exception ex)
                {
                    Console.WriteLine("Bir hata meydana geldi : DB Initialize in IntitializeDatabase.cs : " + ex);
                }
                transaction.Commit();
                Console.WriteLine("Oluşturma başarılı");
            }
            conn.Close();
        }
    }
}
